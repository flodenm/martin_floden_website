\section{Preliminary lecture outline}


\begin{frame}{Lecture outline}
\begin{itemize}
    \item [L1.] \textbf{Course introduction} 
    \small{
    \item[] National accounts, prices and productivity; growth facts; mathematical preliminaries
    \item[] Literature: Jones, chapters 1-3
    }
\end{itemize}

\bigskip

\begin{itemize}
    \item [L2.] \textbf{Economic growth and the Solow model} 
    \small{
    \item[] A model of production; growth accounting; the Solow model
    \item[] Literature: Jones, chapters 3-5
    \item[] Recommended reading: \href{https://www.nber.org/system/files/working_papers/w19895/w19895.pdf}{Gordon (2014, sections 1-3.3)}; \href{https://www.kansascityfed.org/Jackson\%20Hole/documents/9771/GrowthOutlookPanel.pdf}{Jones (2023)}}; \href{https://commission.europa.eu/document/download/97e481fd-2dc3-412d-be4c-f152a8232961_en?filename=The\%20future\%20of\%20European\%20competitiveness\%20\_\%20A\%20competitiveness\%20strategy\%20for\%20Europe.pdf}{Draghi (2024, pages 5-19)}
\end{itemize}
\end{frame}


\begin{frame}{Lecture outline cont'd.}
\begin{itemize}
    \item [L3.] \textbf{More on the Solow model}
    \small{
    \item[] Human capital; finite natural resources; Malthus and the dismal science
    \item[] Literature: Jones, chapter 5; \href{https://eml.berkeley.edu/~jsteinsson/teaching/malthus.pdf}{*Steinsson (2025a)}
    }
\end{itemize}

\bigskip

\begin{itemize}
    \item [L4.] \textbf{Productivity, ideas and long-term economic growth} 
    \small{
    \item[] The AK model; the Romer model; growth accounting; institutions and growth
    \item[] Literature: Jones, chapter 6 
    \item[] Recommended reading: \href{https://www.nobelprize.org/uploads/2018/10/popular-economicsciencesprize2018.pdf}{Royal Swedish Academy of Sciences (2018)}; \href{https://www.nobelprize.org/uploads/2024/10/popular-economicsciencesprize2024.pdf}{Royal Swedish Academy of Sciences (2024)}
    }
\end{itemize}
\end{frame}


\begin{frame}{Lecture outline cont'd.}
\begin{itemize}
    \item [L5.] \textbf{Household behavior}
    \small{
    \item[] Labor supply; the labor market; unemployment; consumption; investment
    \item[] Literature: Jones, chapters 7, 16 and parts of 17; \href{https://eml.berkeley.edu/~jsteinsson/teaching/labor.pdf}{*Steinsson (2025b, sections 1-2 and 4-4.4)}
    }
\end{itemize}

\bigskip

\begin{itemize}
    \item[L6.] \textbf{The short run}
    \small{
    \item[] Economic fluctuations ("business cycles"), the IS curve; monetary policy; the Phillips curve; the IS-MP-PC framework
    \item[] Literature: Jones, chapters 8-12
    \item[] Recommended reading: \href{https://ir.citi.com/p\%2FJ9Uj80\%2BaxALKreeY0Z1vUZImlOZOLlB6g4Rv8kllN5sDCBZyPvGe3ti1Rfq\%2BSM}{Buiter and Rahbari (2015)}; \href{https://cepr.org/voxeu/columns/inflation-covid-demand-or-supply}{Cerrato and Gitti (2023)}
    }
\end{itemize}
\end{frame}


\begin{frame}{Lecture outline cont'd.}
\begin{itemize}
    \item [L7.] \textbf{The short run}
    \small{
    \item[] Propagation of economic fluctuations, multipliers; the AS-AD model; the Taylor rule for monetary policy
    \item[] Literature: Jones, chapters 9-13
    \item[] Recommended reading: 
    \href{https://cepr.org/voxeu/columns/drivers-post-pandemic-inflation}{Giannone and Primiceri (2024)}; \href{https://eml.berkeley.edu/~jsteinsson/papers/Taylor_Rule.pdf}{Nakamura, Riblier and Steinsson (2025)}
    }
\end{itemize}

\bigskip

\begin{itemize}
    \item[L8.] \textbf{Money and money supply}
    \small{
    \item[] Financial markets in the macroeconomy; money and central banking; the time-inconsistency problem of monetary policy
    \item[] Literature: Jones, chapters 8-14
    \item[] Recommended reading: \href{https://www.bankofengland.co.uk/-/media/boe/files/quarterly-bulletin/2014/money-in-the-modern-economy-an-introduction.pdf}{McLeay, Radia and Thomas (2014)}; \href{https://www.nobelprize.org/uploads/2022/10/popular-economicsciencesprize2022-2.pdf}{Royal Swedish Academy of Sciences (2022)}
    }
\end{itemize}
\end{frame}


\begin{frame}{Lecture outline cont'd.}
\begin{itemize}
    \item [L9.] \textbf{Fiscal policy}
    \small{
    \item[] Ricardian equivalence; debt sustainability; fiscal multipliers; automatic stabilizers
    \item[] Literature: Jones, chapters 11, 16 and 18
    \item[] Recommended reading: \href{https://www.brookings.edu/wp-content/uploads/2025/02/20240212_ES_EdelbergHarrisSheiner_Debt_final1.pdf}{Edelberg, Harris and Sheiner (2025)}
    }
\end{itemize}

\bigskip

\begin{itemize}
    \item [L10.] \textbf{Exchange rates and the international monetary system}
    \small{    
    \item[] Purchasing power parity; interest-rate parity; the open-economy IS curve; the trilemma of international macroeconomics; dollar dominance
    \item[] Literature: Jones, chapter 20
    \item[] Recommended reading: \href{https://www.hudsonbaycapital.com/documents/FG/hudsonbay/research/638199_A_Users_Guide_to_Restructuring_the_Global_Trading_System.pdf}{Miran (2024)}; \href{https://www.banque-france.fr/en/governors-interventions/international-monetary-system-unfair}{Bénassy-Quéré (2025)} 
    }
\end{itemize}
\end{frame}



\begin{frame}{Lecture outline cont'd.}
\begin{itemize}
    \item [L11.] \textbf{Trade, tariffs and international capital flows}
    \small{    
    \item[] The current account balance; tariffs
    \item[] Literature: Jones, chapter 19
    \item[] Recommended reading: \href{https://www.bis.org/publ/bisbull110.pdf}{Kohlscheen et al. (2025)}; \href{https://libertystreeteconomics.newyorkfed.org/2025/05/why-does-the-u-s-always-run-a-trade-deficit/}{Klitgaard (2025)}
    }
\end{itemize}

\bigskip

\begin{itemize}
\small{
\item [L12.] \textbf{Review} 
}
\end{itemize}
\end{frame}
